CRUD completo con FastAPI + SQLAlchemy

1. Objetivo de la sesión

Al finalizar esta sesión, el aprendiz será capaz de migrar un CRUD en memoria a persistencia real con SQLAlchemy 2.0: escribir consultas con select(), gestionar la transacción con commit y rollback, devolver objetos ORM a través de esquemas Pydantic con from_attributes, y traducir los errores de integridad de la base a los códigos HTTP correspondientes.

2. Refactorización 2: de la memoria a la base de datos

Esta es la segunda refactorización planeada del bloque, y la más importante. Todo el trabajo de separación hecho hasta ahora se cobra hoy: como los endpoints nunca manipularon los datos directamente, el cambio se concentra en el módulo crud y en los esquemas de respuesta.

Operación

En memoria (hasta ayer)

Con SQLAlchemy 2.0

Listar

[l for l in LIBROS if …]

sesion.scalars(select(Libro).where(…))

Obtener por id

Recorrer la lista

Crear

LISTA.append(diccionario)

sesion.add(objeto) + commit()

Actualizar

diccionario.update(cambios)

setattr sobre el objeto + commit()

Eliminar

LISTA.remove(elemento)

sesion.delete(objeto) + commit()

Confirmar

Nada: ya estaba en la lista

commit() o rollback() ante un fallo

Idea clave

La transacción es lo que la versión en memoria no tenía. Registrar un préstamo son dos escrituras —crear el préstamo y descontar el ejemplar— que deben ocurrir juntas o no ocurrir. Con commit y rollback eso está garantizado; con una lista de Python, un fallo a mitad de camino dejaba el catálogo descuadrado.

Equivalencia con Django REST Framework

El ModelSerializer de DRF hacía de traductor en ambos sentidos: leía el modelo para responder y escribía el modelo al recibir. Aquí esa función se reparte: from_attributes permite a Pydantic leer un objeto ORM para la salida, y el módulo crud construye el objeto ORM a partir del esquema de entrada. La transacción, que en DRF quedaba oculta tras serializer.save(), aquí es explícita: commit y rollback se escriben.

from sqlalchemy import func, select

from sqlalchemy.exc import IntegrityError

from sqlalchemy.orm import Session

from app.errores import ConflictoDeNegocio, RecursoNoEncontrado

from app.models.biblioteca import Autor, Categoria, Libro, Prestamo

def listar(

 sesion: Session,

 categoria: str | None = None,

 anio_min: int | None = None,

 buscar: str | None = None,

 solo_disponibles: bool = False,

 limite: int = 10,

 desplazamiento: int = 0,

) -> list[Libro]:

 consulta = select(Libro)

 if categoria is not None:

 consulta = consulta.join(Libro.categoria).where(Categoria.nombre == categoria)

 if anio_min is not None:

 consulta = consulta.where(Libro.anio_publicacion >= anio_min)

 if buscar is not None:

 consulta = consulta.where(Libro.titulo.ilike(f"%{buscar}%"))

 if solo_disponibles:

 consulta = consulta.where(Libro.ejemplares_disponibles > 0)

 consulta = consulta.order_by(Libro.titulo).offset(desplazamiento).limit(limite)

 return list(sesion.scalars(consulta).unique())

def obtener_o_fallar(sesion: Session, libro_id: int) -> Libro:

 libro = sesion.get(Libro, libro_id)

 if libro is None:

 raise RecursoNoEncontrado("un libro", libro_id)

 return libro

def _validar_relaciones(sesion: Session, autor_id: int, categoria_id: int) -> None:

 if sesion.get(Autor, autor_id) is None:

 raise RecursoNoEncontrado("un autor", autor_id)

 if sesion.get(Categoria, categoria_id) is None:

 raise RecursoNoEncontrado("una categoría", categoria_id)

def crear(sesion: Session, datos: dict) -> Libro:

 _validar_relaciones(sesion, datos["autor_id"], datos["categoria_id"])

 libro = Libro(

 **datos,

 ejemplares_disponibles=datos["ejemplares_totales"],

 )

 sesion.add(libro)

 try:

 sesion.commit()

 except IntegrityError:

 sesion.rollback()

 raise ConflictoDeNegocio(

 f"Ya existe un libro registrado con el ISBN {datos['isbn']}."

 )

 sesion.refresh(libro)

 return libro

def actualizar(sesion: Session, libro_id: int, cambios: dict) -> Libro:

 libro = obtener_o_fallar(sesion, libro_id)

 if "autor_id" in cambios or "categoria_id" in cambios:

 _validar_relaciones(

 sesion,

 cambios.get("autor_id", libro.autor_id),

 cambios.get("categoria_id", libro.categoria_id),

 )

 if "ejemplares_totales" in cambios:

 prestados = libro.ejemplares_totales - libro.ejemplares_disponibles

 libro.ejemplares_disponibles = max(

 cambios["ejemplares_totales"] - prestados, 0

 )

 for campo, valor in cambios.items():

 setattr(libro, campo, valor)

 sesion.commit()

 sesion.refresh(libro)

 return libro

def eliminar(sesion: Session, libro_id: int) -> None:

 libro = obtener_o_fallar(sesion, libro_id)

 activos = sesion.scalar(

 select(func.count())

 .select_from(Prestamo)

 .where(

 Prestamo.libro_id == libro_id,

 Prestamo.fecha_devolucion_real.is_(None),

 )

 )

 if activos:

 raise ConflictoDeNegocio(

 f"El libro tiene {activos} préstamo(s) activo(s); no puede eliminarse."

 )

 sesion.delete(libro)

 sesion.commit()

from datetime import date, timedelta

from sqlalchemy import func, select

from sqlalchemy.exc import IntegrityError

from sqlalchemy.orm import Session

from app.crud import libros as crud_libros

from app.errores import (

 ConflictoDeNegocio,

 LimiteDePrestamosSuperado,

 PrestamoYaDevuelto,

 RecursoNoEncontrado,

 SinEjemplaresDisponibles,

 SocioInactivo,

)

from app.models.biblioteca import Prestamo, Socio

LIMITE_PRESTAMOS_ACTIVOS = 3

def registrar(

 sesion: Session, libro_id: int, socio_id: int, dias_prestamo: int

) -> Prestamo:

 """Las dos escrituras ocurren en la misma transacción: ambas o ninguna."""

 socio = sesion.get(Socio, socio_id)

 if socio is None:

 raise RecursoNoEncontrado("un socio", socio_id)

 if not socio.activo:

 raise SocioInactivo(socio_id)

 activos = sesion.scalar(

 select(func.count())

 .select_from(Prestamo)

 .where(

 Prestamo.socio_id == socio_id,

 Prestamo.fecha_devolucion_real.is_(None),

 )

 )

 if activos >= LIMITE_PRESTAMOS_ACTIVOS:

 raise LimiteDePrestamosSuperado(socio_id, LIMITE_PRESTAMOS_ACTIVOS)

 libro = crud_libros.obtener_o_fallar(sesion, libro_id)

 if libro.ejemplares_disponibles <= 0:

 raise SinEjemplaresDisponibles(libro_id)

 hoy = date.today()

 prestamo = Prestamo(

 libro_id=libro_id,

 socio_id=socio_id,

 fecha_prestamo=hoy,

 fecha_devolucion_esperada=hoy + timedelta(days=dias_prestamo),

 fecha_devolucion_real=None,

 )

 libro.ejemplares_disponibles -= 1

 sesion.add(prestamo)

 try:

 sesion.commit()

 except IntegrityError:

 # Lo impide uq_prestamo_unico: mismo socio, mismo libro, mismo día

 sesion.rollback()

 raise ConflictoDeNegocio(

 "El socio ya tiene registrado hoy un préstamo de ese libro."

 )

 except Exception:

 # Si algo falla, ni se crea el préstamo ni se descuenta el ejemplar

 sesion.rollback()

 raise

 sesion.refresh(prestamo)

 return prestamo

def registrar_devolucion(sesion: Session, prestamo_id: int) -> Prestamo:

 prestamo = sesion.get(Prestamo, prestamo_id)

 if prestamo is None:

 raise RecursoNoEncontrado("un préstamo", prestamo_id)

 if prestamo.fecha_devolucion_real is not None:

 raise PrestamoYaDevuelto(prestamo_id)

 prestamo.fecha_devolucion_real = date.today()

 prestamo.libro.ejemplares_disponibles += 1

 sesion.commit()

 sesion.refresh(prestamo)

 return prestamo

4. Ejemplo detallado: explicación línea por línea

N°

Elemento

Explicación

1

select(Libro)

Construye la consulta sin ejecutarla. Se le van encadenando filtros y solo se materializa al pasarla a la sesión.

2

sesion.scalars(consulta)

Devuelve los objetos Libro. Con execute() se obtendrían filas (tuplas), no instancias del modelo.

3

.unique()

Obligatorio cuando hay relaciones con lazy="joined": el JOIN puede repetir filas y unique() elimina los duplicados.

4

Libro.titulo.ilike(f"%{buscar}%")

Búsqueda sin distinguir mayúsculas. Se construye como expresión de SQLAlchemy, nunca concatenando texto SQL a mano.

5

sesion.get(Libro, libro_id)

Búsqueda por clave primaria. Consulta la caché de identidad de la sesión antes de ir a la base. Sustituye a query.get() de la versión 1.x.

6

sesion.add(libro)

Marca el objeto para insertar. Todavía no hay ningún INSERT: ocurre en el flush o el commit.

7

except IntegrityError: rollback()

La base rechazó el ISBN duplicado. Sin rollback la sesión queda inutilizable y toda operación posterior falla.

8

sesion.refresh(libro)

Recarga el objeto desde la base para obtener el id asignado y los valores por defecto que calculó el motor.

9

setattr(libro, campo, valor)

La actualización se hace sobre el objeto; SQLAlchemy detecta qué atributos cambiaron y genera solo esos UPDATE.

10

select(func.count()).select_from(Prestamo)

Cuenta en la base. Traer todos los préstamos a Python para hacer len() funciona con veinte registros y colapsa con veinte mil.

11

Prestamo.fecha_devolucion_real.is_(None)

En SQL la comparación con nulo se escribe IS NULL. Usar == None funciona por sobrecarga, pero is_() es explícito y no lo marcan los linters.

12

libro.ejemplares_disponibles -= 1 antes del commit

Ambas escrituras quedan en la misma transacción. Si el commit falla, el rollback deshace las dos.

13

from_attributes=True

Permite a Pydantic construir la respuesta leyendo atributos del objeto ORM. Sin esto aparece «Input should be a valid dictionary».

14

autor: AutorResumen anidado

Pydantic navega libro.autor y lo convierte. El esquema decide qué campos del autor se publican: no salen sus otros libros.

15

@computed_field sobre una property

Añade a la respuesta un campo que no existe en la base. El estado se calcula al responder, así que nunca queda obsoleto.

5. Errores comunes y cómo evitarlos

Olvidar el commit: la operación responde 201 y el dato desaparece al cerrar la sesión, porque nunca llegó a la base.

No hacer rollback tras un IntegrityError: la sesión queda en estado inválido y la siguiente operación falla con «This Session's transaction has been rolled back».

Devolver un objeto ORM sin from_attributes en el esquema: «Input should be a valid dictionary or instance of …».

Anidar el esquema completo del autor en la respuesta del libro y que este incluya sus libros: se genera una recursión y una respuesta enorme. Por eso existen los esquemas Resumen.

Omitir .unique() con relaciones joined: SQLAlchemy lanza «The unique() method must be invoked on this Result».

Filtrar o contar en Python después de traer toda la tabla: funciona en clase y no en producción.

Usar query.get(), Query.filter_by() o session.query(): API heredada de SQLAlchemy 1.x. En 2.0 es select() con where().

Crear el préstamo y descontar el ejemplar en dos commits separados: si el segundo falla, el catálogo queda descuadrado.
