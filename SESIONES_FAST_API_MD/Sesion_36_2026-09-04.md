OAuth2 y JWT

Autenticación es quién eres (401); autorización es qué puedes hacer (403). La contraseña se guarda hasheada, el token se firma con una clave que vive en el entorno, y siempre lleva expiración.

SQLAlchemy 2.0

Engine único por proceso, sesión por petición entregada con yield. Modelo ORM y esquema Pydantic son cosas distintas. lazy="joined" evita el problema N+1.

CRUD persistente

select() y where() en lugar de comprensiones de listas; commit y rollback explícitos; from_attributes para responder desde el objeto ORM; IntegrityError traducido a 409.

Idea clave

El cambio de memoria a base de datos tocó el módulo crud y los esquemas de respuesta, pero no las rutas ni los códigos de estado. Ese es el rendimiento de haber separado capas desde agosto: el contrato público de la API no se movió aunque su interior cambiara por completo.

1

La petición no trae token. ¿401 o 403?

401: no se sabe quién es. El 403 exige haber identificado al solicitante.

2

¿Por qué el mismo mensaje para documento inexistente y contraseña incorrecta?

Porque distinguirlos convierte el login en un verificador de qué documentos están registrados.

3

¿Qué pasa si un JWT no lleva exp?

Es válido indefinidamente y no hay forma de retirarlo salvo cambiar la clave secreta, lo que invalidaría todos los tokens.

4

¿Por qué el engine no puede crearse dentro de un endpoint?

Porque abriría un pool de conexiones nuevo por petición y agotaría la base en minutos.

5

Se responde 201 pero el dato no está en la base. ¿Qué falta?

El commit. Sin él la transacción se descarta al cerrar la sesión.

6

¿Qué ocurre si tras un IntegrityError no se hace rollback?

La sesión queda invalidada y toda operación posterior falla con «This Session's transaction has been rolled back».

7

¿Por qué existen los esquemas AutorResumen y CategoriaResumen?

Para cortar el anidamiento: si el libro incluyera el esquema completo del autor, y este sus libros, la respuesta sería recursiva.

8

¿Qué es el problema N+1 y cómo se detecta?

Una consulta por cada elemento de la lista al recorrer una relación. Se detecta con echo=True mirando cuántos SELECT se generan.

9

¿Por qué el estado del préstamo se calcula y no se guarda?

Porque depende de la fecha actual: un valor almacenado quedaría desactualizado sin que nada lo modifique.

from sqlalchemy import func, select

from sqlalchemy.exc import IntegrityError

from sqlalchemy.orm import Session

from app.errores import ConflictoDeNegocio, RecursoNoEncontrado

from app.models.biblioteca import Autor, Libro

def listar(sesion: Session, limite: int = 10, desplazamiento: int = 0) -> list[Autor]:

 consulta = (

 select(Autor).order_by(Autor.nombre).offset(desplazamiento).limit(limite)

 )

 return list(sesion.scalars(consulta))

def obtener_o_fallar(sesion: Session, autor_id: int) -> Autor:

 autor = sesion.get(Autor, autor_id)

 if autor is None:

 raise RecursoNoEncontrado("un autor", autor_id)

 return autor

def contar_libros(sesion: Session, autor_id: int) -> int:

 return sesion.scalar(

 select(func.count()).select_from(Libro).where(Libro.autor_id == autor_id)

 )

def crear(sesion: Session, datos: dict) -> Autor:

 autor = Autor(**datos)

 sesion.add(autor)

 try:

 sesion.commit()

 except IntegrityError:

 sesion.rollback()

 raise ConflictoDeNegocio(

 f"Ya existe un autor llamado {datos['nombre']}."

 )

 sesion.refresh(autor)

 return autor

def eliminar(sesion: Session, autor_id: int) -> None:

 autor = obtener_o_fallar(sesion, autor_id)

 libros = contar_libros(sesion, autor_id)

 if libros:

 raise ConflictoDeNegocio(

 f"El autor tiene {libros} libro(s) en el catálogo; no puede eliminarse."

 )

 sesion.delete(autor)

 sesion.commit()

from typing import Literal

from pydantic import BaseModel, ConfigDict, EmailStr, Field, field_validator

class SocioCrear(BaseModel):

 """La contraseña entra en claro y sale hasheada: nunca se almacena así."""

 documento: str = Field(pattern=r"^\d{6,15}$")

 nombre: str = Field(min_length=5, max_length=120)

 email: EmailStr

 contrasena: str = Field(min_length=8, max_length=128)

 rol: Literal["socio", "bibliotecario"] = "socio"

 @field_validator("contrasena")

 @classmethod

 def suficientemente_fuerte(cls, valor: str) -> str:

 if valor.isalpha() or valor.isdigit():

 raise ValueError(

 "La contraseña debe combinar letras y números."

 )

 return valor

class SocioRespuesta(BaseModel):

 """Ni el documento ni el hash de la contraseña salen en la respuesta."""

 model_config = ConfigDict(from_attributes=True)

 id: int

 nombre: str

 email: EmailStr

 activo: bool

 rol: str

app/routers/socios.py — protegido y persistente

from typing import Annotated

from fastapi import APIRouter, Depends, Path, Query, status

from sqlalchemy import select

from sqlalchemy.exc import IntegrityError

from app.core.seguridad import hashear_contrasena

from app.dependencias import SesionDep, exigir_bibliotecario

from app.errores import ConflictoDeNegocio, RecursoNoEncontrado

from app.models.biblioteca import Socio

from app.schemas.error import DetalleDeError

from app.schemas.socio import SocioCrear, SocioRespuesta

router = APIRouter(

 prefix="/socios",

 tags=["Socios"],

 dependencies=[Depends(exigir_bibliotecario)],

 responses={

 401: {"model": DetalleDeError},

 403: {"model": DetalleDeError},

 },

)

@router.get("", response_model=list[SocioRespuesta], summary="Listar socios")

def listar_socios(

 sesion: SesionDep,

 activo: Annotated[bool | None, Query()] = None,

 limite: Annotated[int, Query(ge=1, le=50)] = 10,

 desplazamiento: Annotated[int, Query(ge=0)] = 0,

):

 consulta = select(Socio)

 if activo is not None:

 consulta = consulta.where(Socio.activo.is_(activo))

 consulta = consulta.order_by(Socio.nombre).offset(desplazamiento).limit(limite)

 return list(sesion.scalars(consulta))

@router.get("/{socio_id}", response_model=SocioRespuesta, summary="Consultar un socio")

def obtener_socio(sesion: SesionDep, socio_id: Annotated[int, Path(ge=1)]):

 socio = sesion.get(Socio, socio_id)

 if socio is None:

 raise RecursoNoEncontrado("un socio", socio_id)

 return socio

@router.post(

 "",

 response_model=SocioRespuesta,

 status_code=status.HTTP_201_CREATED,

 summary="Registrar un socio",

 responses={409: {"model": DetalleDeError}},

)

def crear_socio(sesion: SesionDep, datos: SocioCrear):

 valores = datos.model_dump()

 # La contraseña en claro no llega nunca al modelo

 contrasena = valores.pop("contrasena")

 socio = Socio(**valores, contrasena_hash=hashear_contrasena(contrasena))

 sesion.add(socio)

 try:

 sesion.commit()

 except IntegrityError:

 sesion.rollback()

 raise ConflictoDeNegocio(

 "Ya existe un socio con ese documento o correo electrónico."

 )

 sesion.refresh(socio)

 return socio

4. Errores comunes recopilados de las sesiones anteriores

Dejar la contraseña en el esquema de respuesta o el documento visible para cualquiera: el esquema de salida decide qué se publica, y debe revisarse campo por campo.

Guardar la contraseña sin hashear porque «es un proyecto de clase»: el hábito es lo que se está formando, y es el mismo código que se lleva al trabajo.

Olvidar el commit, o hacerlo sin rollback ante el error: el dato se pierde o la sesión queda inservible.

Migrar el crud pero dejar los endpoints leyendo las listas de app/datos.py: conviven dos fuentes de verdad y los datos dejan de coincidir.

Repetir la comprobación de rol en cada endpoint del router en lugar de declararla una vez en dependencies.

Usar EmailStr sin instalar email-validator: FastAPI lanza «email-validator is not installed» al importar el esquema.

Devolver el objeto ORM sin esquema de respuesta y filtrar así, sin querer, el hash de la contraseña.

Confiar solo en la validación de Pydantic para la unicidad: la restricción debe existir también en la columna de la base.
